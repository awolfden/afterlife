cmake_minimum_required(VERSION 3.16)
if(NOT DEFINED ENV{WATCHER_M5GFX_PATH})
  message(FATAL_ERROR "Set WATCHER_M5GFX_PATH to the audited M5GFX 0.2.26 checkout")
endif()
execute_process(COMMAND git -C "$ENV{WATCHER_M5GFX_PATH}" rev-parse HEAD OUTPUT_VARIABLE GFX_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT GFX_REV STREQUAL "729297d6e3d657ddc1ec5189bac2f2ea68828085")
  message(FATAL_ERROR "Unreviewed M5GFX revision")
endif()
execute_process(COMMAND git -C "$ENV{WATCHER_M5GFX_PATH}" status --porcelain --untracked-files=all OUTPUT_VARIABLE GFX_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT GFX_DIRTY STREQUAL "")
  message(FATAL_ERROR "M5GFX source tree is not clean")
endif()
execute_process(COMMAND git -C "$ENV{IDF_PATH}" rev-parse HEAD OUTPUT_VARIABLE SDK_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND git -C "$ENV{IDF_PATH}" status --porcelain --untracked-files=all OUTPUT_VARIABLE SDK_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT SDK_REV STREQUAL "735507283d5b2f9fb363a1901172dbd9e847945d" OR NOT SDK_DIRTY STREQUAL "")
  message(FATAL_ERROR "Expected clean audited ESP-IDF 5.5.4 source")
endif()
set(ENV{IDF_COMPONENT_MANAGER} "0")
option(WATCHER_LAN "RAM-only paired LAN transport (experimental; no provisioning persistence)" OFF)
set(COMPONENTS main)
if(WATCHER_LAN)
  list(APPEND COMPONENTS watcher_lan)
endif()
set(EXTRA_COMPONENT_DIRS "$ENV{WATCHER_M5GFX_PATH}")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(watcher_pet)
